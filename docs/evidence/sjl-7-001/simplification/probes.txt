# classes.js:
const rows = [];
for (const [source, text] of [['[\\p{Lu}]', '\u{1D400}'], ['^[\\p{Lu}]$', '\u{16EA0}'], ['^\\p{Lu}$', '\u{16EA0}'],
    ['^[\\p{Ll}]$', '\u{16EBB}'], ['^[\\p{L}]$', '\u{105C0}'], ['^\\p{L}$', '\u{105C0}'], ['^[\\p{gc=Lu}x]$', '\u{1D400}']]) {
  let answer;
  try { answer = new RegExp(source, 'u').test(text); } catch (e) { answer = 'threw ' + e.name; }
  rows.push('/' + source + '/u on U+' + text.codePointAt(0).toString(16).toUpperCase() + ': ' + answer);
}
console.log(rows.join('\n'));
